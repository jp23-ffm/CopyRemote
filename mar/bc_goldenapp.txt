# monitor/signals.py
TRUSTED_PROXY_COUNT = 2  # load balancer + nginx

def _get_ip(request):
    x_forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR')
    if x_forwarded_for:
        parts = [p.strip() for p in x_forwarded_for.split(',') if p.strip()]
        if len(parts) >= TRUSTED_PROXY_COUNT:
            return parts[-TRUSTED_PROXY_COUNT]
        return parts[0] if parts else request.META.get('REMOTE_ADDR')
    return request.META.get('REMOTE_ADDR')

# businesscontinuity/management/commands/purge_expired_cache.py
from django.core.management.base import BaseCommand
from django.utils import timezone
from django.db import connection

class Command(BaseCommand):
    def handle(self, *args, **options):
        with connection.cursor() as cur:
            cur.execute(
                "DELETE FROM inventory_listbox_cache WHERE expires < %s",
                [timezone.now()]
            )
            self.stdout.write(f"{cur.rowcount} entrées expirées supprimées")
			


'longcache': {
    'BACKEND': 'django.core.cache.backends.db.DatabaseCache',
    'LOCATION': 'inventory_listbox_cache',
    'OPTIONS': {
        'MAX_ENTRIES': 300,       # déclenche le nettoyage au-delà
        'CULL_FREQUENCY': 3,      # purge 1/3 des entrées à chaque cull
    }
}
